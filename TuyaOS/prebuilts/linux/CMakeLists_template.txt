function(include_subdirectories root_dir)
    if(IS_DIRECTORY ${root_dir})
        include_directories(${root_dir})
    endif()
    file(GLOB ALL_SUB RELATIVE ${root_dir} ${root_dir}/*)
    foreach(sub ${ALL_SUB})
        if(IS_DIRECTORY ${root_dir}/${sub})
            # 检查是否是 lib_tflm 目录
            if("lib_tflm" MATCHES ${sub})
                # 如果是 lib_tflm，只添加 lib_tflm 和特定的子目录
                include_directories(${root_dir}/${sub})
                include_directories(${root_dir}/${sub}/third_party/flatbuffers/include)
                include_directories(${root_dir}/${sub}/third_party/gemmlowp)
            elseif(NOT (".git" MATCHES ${sub} OR "output" MATCHES ${sub}))
                # 如果不是 lib_tflm，递归地包含其他子目录
                include_subdirectories(${root_dir}/${sub})
            endif()
        endif()
    endforeach()
endfunction()

function(add_appimg target ldscript)
    set(gen_ldscript ${target}_ldscript)
    set(target_app_image_file ${out_hex_dir}/${APP_NAME}.bin)
    set(target_map_file ${out_hex_dir}/${target}.map)
    set(target_img_file ${out_hex_dir}/${target}.binpkg)

    add_executable(${target} ${ARGN})
    target_link_directories(${target} PRIVATE ${PROJECT_ROOT_DIR}/libs)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${out_hex_dir})

    # 设置链接器脚本和其他链接选项
    target_link_options(${target} PRIVATE
        -T ${ldscript}
        -Wl,-Map=${target_map_file}
        -Wl,--gc-sections
        -Wl,--no-undefined
        -Wl,--no-print-map-discarded
        -Wl,--no-whole-archive
    )
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${target}> ${target_app_image_file}
    )
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${cmd_mksimage} -M
        -input ${PROJECT_ROOT_DIR}/sysfw/ap_bootloader.bin -addrname BL_PKGIMG_LNA -flashsize BOOTLOADER_PKGIMG_LIMIT_SIZE
        -input ${target_app_image_file}  -addrname AP_PKGIMG_LNA -flashsize AP_PKGIMG_LIMIT_SIZE
        -input ${PROJECT_ROOT_DIR}/sysfw/cp-demo-flash.bin -addrname CP_PKGIMG_LNA -flashsize CP_PKGIMG_LIMIT_SIZE
        -pkgmode 1 -banoldtool 1
        -def ${PROJECT_ROOT_DIR}/sysfw/mem_map.txt
        -outfile ${target_img_file}
        DEPENDS ${APP_NAME}
        COMMENT "Creating binpkg file"
    )
endfunction()
